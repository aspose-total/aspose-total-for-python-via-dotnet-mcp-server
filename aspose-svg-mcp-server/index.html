Aspose.SVG MCP Server

SVG

Aspose.SVG for Python via .NET provides server‑side tools to create, edit, query, and convert SVG files within a shared files directory.

Configure this server Official Aspose.SVG page View source folder
Setup

Configure Aspose.SVG in your AI client

Add one entry to your MCP host config — only the aspose-svg key changes across clients.

{
  "mcpServers": {
    "aspose-svg": {
      "command": "/path/to/servers/svg/.venv/bin/python",
      "args": ["/path/to/servers/svg/server.py"],
      "env": {
        "ASPOSE_FILES_PATH": "/home/user/asposefiles",
        "ASPOSE_LICENSE_FILE": "/home/user/asposefiles/Aspose.License.File.Name.lic"
      }
    }
  }
}
{
  "servers": {
    "aspose-svg": {
      "type": "stdio",
      "command": "/path/to/servers/svg/.venv/bin/python",
      "args": ["/path/to/servers/svg/server.py"],
      "env": {
        "ASPOSE_FILES_PATH": "/home/user/asposefiles",
        "ASPOSE_LICENSE_FILE": "/home/user/asposefiles/Aspose.License.File.Name.lic"
      }
    }
  }
}
